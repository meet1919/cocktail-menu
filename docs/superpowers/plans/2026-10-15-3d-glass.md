# 3D Glass Stage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the pinned SVG glass with a real-time 3D scene (refractive glass, physics-driven ingredients, drag to rotate, bar environment), keeping the SVG stage as an automatic fallback.

**Architecture:** New ES modules in `js/stage3d/`, loaded by a dynamic `import()` from the existing classic scripts. A `Stage3D` object (`kind: '3d'`) mirrors the SVG stage's role. `scroll.js` gets a 3D branch that feeds each section's scroll progress to a `director`, which spawns and removes physics bodies. Three.js renders, cannon-es simulates, GSAP (already a global) handles tweens.

**Tech Stack:** Three.js 0.170.0 and cannon-es 0.20.0 (import map, jsDelivr), GSAP 3.12 + ScrollTrigger (existing), vanilla ES modules, no build step.

**Spec:** `docs/superpowers/specs/2026-10-15-3d-glass-design.md`

## Global Constraints

- Units: 1 unit = 1 cm. Fill maximum = `0.85` of the inner height.
- Pinned versions: `three@0.170.0`, `cannon-es@0.20.0`. Libraries load only from `cdn.jsdelivr.net`.
- No image or 3D model files: all textures are drawn on a canvas, all geometry is generated.
- Glass sizes: highball H 14 (r bottom 3.2, top 3.6, wall 0.25); rocks H 8.5 (r 3.6 to 4.2, wall 0.3, base 1.2); coupe H 14 (R 5.5, bowl 4.5, wall 0.2).
- At most 24 physics bodies at once.
- Scroll range per drink unchanged: `start: 'top 50%'`, `end: 'bottom 80%'`. With `n` ingredients: `T = n + 1.6`, `x = progress × T`; ingredient `j` present when `x ≥ j + 0.3`; `level = 0.85 × clamp((x − 0.5) / n, 0, 1)`; effect on when `x ≥ n + 0.4`.
- Fallback to SVG when: no WebGL2, `prefers-reduced-motion`, `hardwareConcurrency < 4` or `deviceMemory < 4`, `?3d=0`, or import/setup fails or exceeds 6 s. `?3d=1` forces 3D.
- Wheel and vertical touch swipes must always scroll the page (`touch-action: pan-y`, no wheel listener).
- Quality drop: if average frame time over 2 s exceeds 28 ms (after a 3 s warm-up), set pixel ratio 1 and replace transmission materials with simple transparent ones, once.
- Existing behaviour (names, taglines, preloader, SVG fallback) must not change.

## Testing approach

Visual and interactive work, so every task is verified in the in-app browser (`python -m http.server 5188` in the project folder, then `http://localhost:5188/?v=<n>`; change `v` each load to dodge the cache). The console must show **0 errors** after every task. The browser pane pauses animation while hidden, so take a screenshot first to bring it forward. Tuning constants (light intensity, gravity, buoyancy) are starting values and are adjusted by eye in the task that introduces them.

```js
(function () {
  const param = new URLSearchParams(location.search).get('3d');

  function webgl2() {
    try { return !!document.createElement('canvas').getContext('webgl2'); } catch (e) { return false; }
  }

  function allowed(reduced) {
    if (param === '0') return false;
    if (param === '1') return webgl2();
    if (reduced) return false;
    if (navigator.hardwareConcurrency && navigator.hardwareConcurrency < 4) return false;
    if (navigator.deviceMemory && navigator.deviceMemory < 4) return false;
    return webgl2();
  }

  Bar.try3D = async function (container, opts) {
    if (!allowed(opts && opts.reduced)) return null;
    let timer;
    try {
      const timeout = new Promise((_, rej) => { timer = setTimeout(() => rej(new Error('3D load timed out')), 6000); });
      const run = import('./stage3d/index.js').then(m => m.createStage3D(container, Bar.COCKTAILS));
      const stage = await Promise.race([run, timeout]);
      return stage;
    } catch (err) {
      console.warn('3D stage unavailable, using the SVG glass.', err);
      const c = container.querySelector('canvas.stage3d');
      if (c) c.remove();
      return null;
    } finally {
      clearTimeout(timer);
    }
  };
})();
```

```js
import * as THREE from 'three';
import { RoomEnvironment } from 'three/addons/environments/RoomEnvironment.js';
import { quality } from './quality.js';

function radialTexture(stops) {
  const c = document.createElement('canvas');
  c.width = c.height = 128;
  const g = c.getContext('2d');
  const gr = g.createRadialGradient(64, 64, 0, 64, 64, 64);
  stops.forEach(([o, col]) => gr.addColorStop(o, col));
  g.fillStyle = gr;
  g.fillRect(0, 0, 128, 128);
  return new THREE.CanvasTexture(c);
}

export function createScene(canvas, container) {
  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true, powerPreference: 'high-performance' });
  const maxDpr = innerWidth < 768 ? 1.5 : 2;
  renderer.setPixelRatio(Math.min(devicePixelRatio || 1, maxDpr));
  renderer.setClearColor(0x000000, 0);
  renderer.toneMapping = THREE.ACESFilmicToneMapping;
  renderer.toneMappingExposure = 1.05;
  renderer.localClippingEnabled = true;

  const scene = new THREE.Scene();
  const pmrem = new THREE.PMREMGenerator(renderer);
  scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;
  scene.environmentIntensity = 0.65;

  const camera = new THREE.PerspectiveCamera(35, 1, 1, 300);

  const key = new THREE.SpotLight(0xffffff, 1800, 90, 0.55, 0.7, 2);
  key.position.set(14, 26, 16);
  key.target.position.set(0, 6, 0);
  const rim = new THREE.DirectionalLight(0xffffff, 1.2);
  rim.position.set(-10, 12, -14);
  scene.add(key, key.target, rim, new THREE.AmbientLight(0xffffff, 0.15));

  // bar counter that fades out toward its edge, plus a soft contact shadow
  const counter = new THREE.Mesh(
    new THREE.CircleGeometry(45, 64),
    new THREE.MeshStandardMaterial({
      color: 0x15110e, roughness: 0.38, metalness: 0.15, transparent: true,
      alphaMap: radialTexture([[0, '#fff'], [0.55, '#fff'], [1, '#000']])
    })
  );
  counter.rotation.x = -Math.PI / 2;
  const shadow = new THREE.Mesh(
    new THREE.PlaneGeometry(18, 18),
    new THREE.MeshBasicMaterial({ map: radialTexture([[0, 'rgba(0,0,0,.6)'], [1, 'rgba(0,0,0,0)']]), transparent: true, depthWrite: false })
  );
  shadow.rotation.x = -Math.PI / 2;
  shadow.position.y = 0.02;
  scene.add(counter, shadow);

  // out-of-focus warm lights on a ring around the glass
  const glow = radialTexture([[0, 'rgba(255,255,255,1)'], [1, 'rgba(255,255,255,0)']]);
  const bokeh = [];
  for (let i = 0; i < 30; i++) {
    const mat = new THREE.SpriteMaterial({
      map: glow, color: new THREE.Color().setHSL(0.07 + Math.random() * 0.05, 0.9, 0.55),
      transparent: true, opacity: 0.1 + Math.random() * 0.18, blending: THREE.AdditiveBlending, depthWrite: false
    });
    const s = new THREE.Sprite(mat);
    const a = Math.random() * Math.PI * 2, r = 42 + Math.random() * 14, sz = 4 + Math.random() * 9;
    s.position.set(Math.cos(a) * r, 4 + Math.random() * 30, Math.sin(a) * r);
    s.scale.set(sz, sz, 1);
    s.userData = { y0: s.position.y, ph: Math.random() * 6.28, sp: 0.15 + Math.random() * 0.2 };
    scene.add(s);
    bokeh.push(s);
  }

  // loop, gated on visibility
  const frameCbs = [], resizeCbs = [];
  let running = false, visible = true, last = 0, elapsed = 0, ema = 16, windowT = 0;
  function loop(now) {
    const dt = Math.min(0.05, (now - last) / 1000 || 0.016);
    last = now;
    elapsed += dt;
    bokeh.forEach(s => { s.position.y = s.userData.y0 + Math.sin(elapsed * s.userData.sp + s.userData.ph) * 1.5; });
    frameCbs.forEach(f => f(dt));
    renderer.render(scene, camera);
    if (!quality.degraded && elapsed > 3) {
      ema = ema * 0.95 + dt * 1000 * 0.05;
      windowT += dt;
      if (windowT > 2) { if (ema > 28) degrade(); windowT = 0; }
    }
  }
  function updateRun() {
    const should = visible && !document.hidden;
    if (should && !running) { running = true; last = performance.now(); renderer.setAnimationLoop(loop); }
    else if (!should && running) { running = false; renderer.setAnimationLoop(null); }
  }
  new IntersectionObserver(es => { visible = es[0].isIntersecting; updateRun(); }).observe(container);
  document.addEventListener('visibilitychange', updateRun);

  function resize() {
    const w = container.clientWidth, h = container.clientHeight;
    if (!w || !h) return;
    renderer.setSize(w, h, false);
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
    resizeCbs.forEach(f => f());
  }
  new ResizeObserver(resize).observe(container);
  resize();
  updateRun();

  function degrade() {
    if (quality.degraded) return;
    quality.degraded = true;
    renderer.setPixelRatio(1);
    resize();
    scene.traverse(o => {
      [].concat(o.material || []).forEach(m => {
        if (m.transmission > 0) { m.transmission = 0; m.transparent = true; m.opacity = Math.min(m.opacity, 0.3); m.needsUpdate = true; }
      });
    });
  }

  return {
    THREE, renderer, scene, camera, key, rim,
    onFrame: f => frameCbs.push(f),
    onResize: f => resizeCbs.push(f),
    degrade,
    dispose() { renderer.setAnimationLoop(null); renderer.dispose(); }
  };
}
```

```js
import * as THREE from 'three';
import { quality } from './quality.js';

const SPECS = {
  highball: { H: 14, rb: 3.2, rt: 3.6, w: 0.25, base: 0.9 },
  rocks: { H: 8.5, rb: 3.6, rt: 4.2, w: 0.3, base: 1.2 },
  coupe: { H: 14, R: 5.5, w: 0.2, bowl: 4.5 }
};

function straight(s) {
  const rbi = s.rb - s.w, rti = s.rt - s.w;
  const pts = [[0, 0], [s.rb - 0.25, 0], [s.rb, 0.2], [s.rt, s.H], [rti, s.H], [rbi, s.base], [0, s.base]];
  const inner = {
    baseY: s.base, topY: s.H,
    radiusAt: y => rbi + (rti - rbi) * Math.min(1, Math.max(0, (y - s.base) / (s.H - s.base)))
  };
  return { pts, inner, size: { h: s.H, w: s.rt * 2 } };
}

function coupe(s) {
  const { R, w, bowl, H } = s;
  const yb = H - bowl, Ri = R - w, ybi = yb + w, di = H - ybi;
  const pts = [[0, 0], [3.2, 0], [3.3, 0.2], [1.1, 0.55], [0.38, 1.1], [0.38, yb - 0.2]];
  const N = 18;
  for (let i = 0; i <= N; i++) { const r = 0.38 + (R - 0.38) * i / N; pts.push([r, yb + bowl * Math.pow(r / R, 2)]); }
  pts.push([Ri, H]);
  for (let i = N; i >= 0; i--) { const r = Ri * i / N; pts.push([r, ybi + di * Math.pow(r / Ri, 2)]); }
  const inner = {
    baseY: ybi, topY: H,
    radiusAt: y => (y <= ybi ? 0 : Ri * Math.sqrt(Math.min(1, (y - ybi) / di)))
  };
  return { pts, inner, size: { h: H, w: R * 2 } };
}

export function buildGlass(type) {
  const spec = SPECS[type];
  const { pts, inner, size } = type === 'coupe' ? coupe(spec) : straight(spec);
  const geo = new THREE.LatheGeometry(pts.map(p => new THREE.Vector2(p[0], p[1])), 72);
  const mat = new THREE.MeshPhysicalMaterial({
    color: 0xffffff, metalness: 0, roughness: 0.04, ior: 1.5, thickness: 0.6,
    transmission: quality.degraded ? 0 : 1, transparent: true,
    opacity: quality.degraded ? 0.28 : 1, side: THREE.DoubleSide, envMapIntensity: 1.2
  });
  const group = new THREE.Group();
  group.add(new THREE.Mesh(geo, mat));

  const hit = new THREE.Mesh(
    new THREE.CylinderGeometry(size.w / 2, size.w / 2, size.h, 16),
    new THREE.MeshBasicMaterial({ visible: false })
  );
  hit.position.y = size.h / 2;
  group.add(hit);

  return { group, inner, hit, size, materials: [mat] };
}
```

```js
import * as THREE from 'three';

const FOV = 35 * Math.PI / 180;
const EL_MIN = 5 * Math.PI / 180, EL_MAX = 40 * Math.PI / 180;

export function createControls({ camera, canvas, getHit, onNudge, onResize }) {
  const st = { az: 0.5, el: 0.35, vel: 0, radius: 30, ty: 6, tRadius: 30, tTy: 6, idle: 0 };
  let framing = { h: 14, w: 7 };
  let dragging = false, sx = 0, sy = 0, lx = 0, ly = 0, t0 = 0, moved = 0;
  const ray = new THREE.Raycaster(), ndc = new THREE.Vector2();

  function pick(e) {
    const r = canvas.getBoundingClientRect();
    ndc.set(((e.clientX - r.left) / r.width) * 2 - 1, -((e.clientY - r.top) / r.height) * 2 + 1);
    ray.setFromCamera(ndc, camera);
    const hit = getHit();
    return !!hit && ray.intersectObject(hit, false).length > 0;
  }

  canvas.addEventListener('pointerdown', e => {
    if (!pick(e)) return;
    dragging = true; moved = 0; sx = lx = e.clientX; sy = ly = e.clientY; t0 = performance.now();
    st.vel = 0; st.idle = 0;
    canvas.setPointerCapture(e.pointerId);
  });
  canvas.addEventListener('pointermove', e => {
    if (!dragging) { canvas.style.cursor = pick(e) ? 'grab' : ''; return; }
    const dx = e.clientX - lx, dy = e.clientY - ly;
    lx = e.clientX; ly = e.clientY;
    moved = Math.max(moved, Math.hypot(e.clientX - sx, e.clientY - sy));
    st.az -= dx * 0.008;
    st.vel = -dx * 0.008 * 60;
    st.el = Math.min(EL_MAX, Math.max(EL_MIN, st.el + dy * 0.004));
    st.idle = 0;
  });
  function end() {
    if (!dragging) return;
    dragging = false;
    if (moved < 6 && performance.now() - t0 < 250) { st.vel = 0; onNudge(); }
  }
  canvas.addEventListener('pointerup', end);
  canvas.addEventListener('pointercancel', () => { dragging = false; });

  function refit() {
    const extentV = Math.max(framing.h / 0.72, (framing.w / 0.8) / camera.aspect);
    st.tRadius = extentV / (2 * Math.tan(FOV / 2));
    st.tTy = framing.h * 0.45;
  }
  onResize(refit);

  function update(dt) {
    if (!dragging) {
      st.az += st.vel * dt;
      st.vel *= Math.pow(0.08, dt);
      st.idle += dt;
      if (st.idle > 3 && Math.abs(st.vel) < 0.05) st.az += (0.4 * Math.PI / 180) * dt;
    }
    const k = 1 - Math.exp(-6 * dt);
    st.radius += (st.tRadius - st.radius) * k;
    st.ty += (st.tTy - st.ty) * k;
    camera.position.set(
      st.radius * Math.cos(st.el) * Math.sin(st.az),
      st.ty + st.radius * Math.sin(st.el),
      st.radius * Math.cos(st.el) * Math.cos(st.az)
    );
    camera.lookAt(0, st.ty, 0);
  }

  return {
    update,
    setFraming(height, width, instant) {
      framing = { h: height, w: width };
      refit();
      if (instant) { st.radius = st.tRadius; st.ty = st.tTy; }
    },
    get azimuth() { return st.az; }
  };
}
```

```js
import * as CANNON from 'cannon-es';

const G = -300;           // cm/s^2, stronger than real gravity so drops feel snappy
const BANDS = 6, SEGMENTS = 20, THICK = 1.0;

export function createPhysics({ onSplash }) {
  const world = new CANNON.World({ gravity: new CANNON.Vec3(0, G, 0) });
  world.defaultContactMaterial.friction = 0.25;
  world.defaultContactMaterial.restitution = 0.2;
  let walls = [];
  const items = new Set(), dying = new Set();

  function setGlass(inner, rOuter) {
    walls.forEach(w => world.removeBody(w));
    walls = [];
    const floor = new CANNON.Body({ mass: 0, shape: new CANNON.Box(new CANNON.Vec3(rOuter, 0.5, rOuter)), position: new CANNON.Vec3(0, inner.baseY - 0.5, 0) });
    world.addBody(floor); walls.push(floor);
    const span = inner.topY - inner.baseY;
    for (let k = 0; k < BANDS; k++) {
      const y0 = inner.baseY + span * k / BANDS, y1 = inner.baseY + span * (k + 1) / BANDS;
      const r0 = inner.radiusAt(y0), r1 = inner.radiusAt(y1);
      const rm = (r0 + r1) / 2, ym = (y0 + y1) / 2;
      const len = Math.hypot(r1 - r0, y1 - y0), tilt = Math.atan2(r1 - r0, y1 - y0);
      const half = new CANNON.Vec3(THICK / 2, len / 2 + 0.15, Math.PI * rm / SEGMENTS * 1.15 + 0.05);
      const rc = rm + (THICK / 2) * Math.cos(tilt), yc = ym - (THICK / 2) * Math.sin(tilt);
      const qz = new CANNON.Quaternion(); qz.setFromAxisAngle(new CANNON.Vec3(0, 0, 1), -tilt);
      for (let m = 0; m < SEGMENTS; m++) {
        const phi = (m / SEGMENTS) * Math.PI * 2;
        const b = new CANNON.Body({ mass: 0, shape: new CANNON.Box(half) });
        b.position.set(rc * Math.cos(phi), yc, -rc * Math.sin(phi));
        const qy = new CANNON.Quaternion(); qy.setFromAxisAngle(new CANNON.Vec3(0, 1, 0), phi);
        qy.mult(qz, b.quaternion);
        world.addBody(b); walls.push(b);
      }
    }
  }

  function add(def, mesh, pos) {
    const shape = def.shape === 'sphere' ? new CANNON.Sphere(def.radius) : new CANNON.Box(new CANNON.Vec3(...def.half));
    const body = new CANNON.Body({ mass: def.mass, shape, position: new CANNON.Vec3(pos.x, pos.y, pos.z), allowSleep: false });
    body.quaternion.setFromEuler(Math.random() * 6.28, Math.random() * 6.28, Math.random() * 6.28);
    body.angularVelocity.set((Math.random() - 0.5) * 6, (Math.random() - 0.5) * 6, (Math.random() - 0.5) * 6);
    world.addBody(body);
    const item = { body, mesh, buoy: def.buoy, inside: false, baseScale: mesh.scale.clone(), dyingT: 0 };
    items.add(item);
    return item;
  }

  function remove(item) {
    if (!items.has(item)) return;
    items.delete(item);
    world.removeBody(item.body);
    dying.add(item);
  }

  function clear() {
    items.forEach(i => { world.removeBody(i.body); if (i.mesh.parent) i.mesh.parent.remove(i.mesh); });
    dying.forEach(i => { if (i.mesh.parent) i.mesh.parent.remove(i.mesh); });
    items.clear(); dying.clear();
  }

  function nudge() {
    items.forEach(i => {
      i.body.applyImpulse(new CANNON.Vec3((Math.random() - 0.5) * 60 * i.body.mass, (40 + Math.random() * 40) * i.body.mass, (Math.random() - 0.5) * 60 * i.body.mass));
    });
  }

  function step(dt, surfaceY) {
    items.forEach(it => {
      const b = it.body, depth = surfaceY - b.position.y;
      if (depth > 0) {
        const c = it.buoy, f = Math.min(1, depth / c.depth);
        b.applyForce(new CANNON.Vec3(0, -G * b.mass * c.ratio * f, 0));
        b.linearDamping = 0.85; b.angularDamping = 0.85;
        if (!it.inside) {
          it.inside = true;
          if (b.velocity.y < -4) onSplash(b.position.x, b.position.z, Math.min(1.2, -b.velocity.y / 60));
        }
      } else {
        b.linearDamping = 0.02; b.angularDamping = 0.02; it.inside = false;
      }
    });
    world.step(1 / 60, dt, 3);
    items.forEach(it => { it.mesh.position.copy(it.body.position); it.mesh.quaternion.copy(it.body.quaternion); });
    dying.forEach(it => {
      it.dyingT += dt;
      const s = Math.max(0, 1 - it.dyingT / 0.25);
      it.mesh.scale.copy(it.baseScale).multiplyScalar(s);
      if (s <= 0) { if (it.mesh.parent) it.mesh.parent.remove(it.mesh); dying.delete(it); }
    });
  }

  return { setGlass, add, remove, clear, nudge, step, count: () => items.size };
}
```

```js
import * as THREE from 'three';

const MAX_RIPPLES = 6;

export function createLiquid(inner, colorHex) {
  const full = new THREE.Color(colorHex);
  const pale = full.clone().lerp(new THREE.Color(0xffffff), 0.55);
  const group = new THREE.Group();
  const span = inner.topY - inner.baseY;
  const plane = new THREE.Plane(new THREE.Vector3(0, -1, 0), inner.baseY);

  // volume: inner profile, clipped at the current level
  const pts = [new THREE.Vector2(0, inner.baseY)];
  for (let i = 0; i <= 24; i++) {
    const y = inner.baseY + span * i / 24;
    pts.push(new THREE.Vector2(inner.radiusAt(y) * 0.97, y));
  }
  const mat = new THREE.MeshPhysicalMaterial({
    color: pale.clone(), roughness: 0.05, transparent: true, opacity: 0.82,
    side: THREE.DoubleSide, clippingPlanes: [plane], envMapIntensity: 0.9
  });
  const volume = new THREE.Mesh(new THREE.LatheGeometry(pts, 48), mat);
  group.add(volume);

  // surface: concentric rings so ripples can displace it
  const sGeo = new THREE.RingGeometry(0.001, 1, 48, 12);
  sGeo.rotateX(-Math.PI / 2);
  const base = Float32Array.from(sGeo.attributes.position.array);
  const surface = new THREE.Mesh(sGeo, new THREE.MeshPhysicalMaterial({
    color: pale.clone(), roughness: 0.04, clearcoat: 1, transparent: true, opacity: 0.88, side: THREE.DoubleSide
  }));
  group.add(surface);

  // foam layer
  const foam = new THREE.Mesh(new THREE.CylinderGeometry(1, 1, 0.5, 48), new THREE.MeshStandardMaterial({ color: 0xf6f1e1, roughness: 0.9 }));
  foam.scale.y = 0; foam.visible = false;
  group.add(foam);

  // bubbles
  const COUNT = 40;
  const bubbles = new THREE.InstancedMesh(new THREE.SphereGeometry(0.12, 8, 8), new THREE.MeshBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.6 }), COUNT);
  const bs = Array.from({ length: COUNT }, () => ({ a: Math.random() * 6.28, r: Math.random(), y: 0, v: 2 + Math.random() * 3 }));
  bubbles.visible = false;
  group.add(bubbles);
  let bubblesOn = false;

  const ripples = [];
  const st = { level: 0, vel: 0, target: 0 };
  const dummy = new THREE.Object3D();

  const yOf = f => inner.baseY + f * span;
  const surfaceY = () => yOf(st.level);

  function setLevel(f) { st.target = f; }
  function setTint(t) { mat.color.copy(pale).lerp(full, Math.min(1, Math.max(0, t))); surface.material.color.copy(mat.color); }
  function ripple(x, z, strength) {
    ripples.push({ x, z, a: strength, age: 0 });
    if (ripples.length > MAX_RIPPLES) ripples.shift();
  }

  function update(dt) {
    // critically damped spring toward the target level
    const w = 6;
    st.vel += (w * w * (st.target - st.level) - 2 * w * st.vel) * dt;
    st.level += st.vel * dt;
    const y = surfaceY();
    plane.constant = y;
    const R = Math.max(0.01, inner.radiusAt(Math.max(y, inner.baseY + 0.01)) * 0.97);
    surface.position.y = y;
    surface.scale.set(R, 1, R);
    surface.visible = st.level > 0.004;
    foam.position.y = y + 0.25;
    foam.scale.x = foam.scale.z = R * 0.98;

    // ripples
    const pos = surface.geometry.attributes.position;
    ripples.forEach(r => { r.age += dt; });
    for (let i = ripples.length - 1; i >= 0; i--) if (ripples[i].age > 1.2) ripples.splice(i, 1);
    for (let i = 0; i < pos.count; i++) {
      const ux = base[i * 3], uz = base[i * 3 + 2];
      let h = 0;
      for (const r of ripples) {
        const d = Math.hypot(ux * R - r.x, uz * R - r.z);
        h += 0.25 * r.a * Math.exp(-r.age * 2.4) * Math.sin(d * 3.5 - r.age * 9) * Math.exp(-d * 0.25);
      }
      pos.setY(i, h);
    }
    pos.needsUpdate = true;
    if (ripples.length) surface.geometry.computeVertexNormals();

    if (bubblesOn) {
      for (let i = 0; i < COUNT; i++) {
        const b = bs[i];
        b.y += b.v * dt;
        if (b.y > y - inner.baseY - 0.2) { b.y = 0; b.a = Math.random() * 6.28; b.r = Math.random(); }
        const yy = inner.baseY + 0.3 + b.y;
        const rr = inner.radiusAt(yy) * 0.8 * b.r;
        dummy.position.set(Math.cos(b.a) * rr, yy, Math.sin(b.a) * rr);
        dummy.updateMatrix();
        bubbles.setMatrixAt(i, dummy.matrix);
      }
      bubbles.instanceMatrix.needsUpdate = true;
    }
  }

  return {
    group, setLevel, setTint, surfaceY, ripple, update,
    setFoam(on) {
      foam.visible = true;
      if (window.gsap) gsap.to(foam.scale, { y: on ? 1 : 0, duration: 0.6, overwrite: 'auto', onComplete: () => { foam.visible = on; } });
      else { foam.scale.y = on ? 1 : 0; foam.visible = on; }
    },
    setBubbles(on) { bubblesOn = on; bubbles.visible = on; }
  };
}
```

```js
  api.setActive = function (i, instant) {

    if (prev) {
      clearDrink(prev);
      gsap.to(prev.glass.group.scale, { x: 0.85, y: 0.85, z: 0.85, duration: instant ? 0 : 0.4, overwrite: 'auto' });
      const o = { v: 1 };
      gsap.to(o, { v: 0, duration: instant ? 0 : 0.4, onUpdate: () => setOpacity(prev, o.v) });
    }
    cur = i;
    const d = dr[i];
    clearDrink(d);
    physics.setGlass(d.glass.inner, d.glass.size.w / 2);
    controls.setFraming(d.glass.size.h, d.glass.size.w, !!instant);
    d.glass.group.scale.set(0.9, 0.9, 0.9);
    gsap.to(d.glass.group.scale, { x: 1, y: 1, z: 1, duration: instant ? 0 : 0.5, overwrite: 'auto' });
    const o = { v: 0 };
    setOpacity(d, instant ? 1 : 0);
    gsap.to(o, { v: 1, duration: instant ? 0 : 0.5, onUpdate: () => setOpacity(d, o.v) });
    const col = new THREE.Color(d.c.color);
    gsap.to(key.color, { r: col.r * 0.6 + 0.4, g: col.g * 0.6 + 0.4, b: col.b * 0.6 + 0.4, duration: 0.8, overwrite: 'auto' });
  };

  api.update = function (i, p) {
    if (i !== cur) return;
    const d = dr[i], ings = d.c.ingredients, n = ings.length;
    const x = p * (n + 1.6);
    for (let j = 0; j < n; j++) {
      const want = x >= j + 0.3, has = d.spawned.has(j);
      if (want && !has) {
        d.spawned.set(j, null);
        queue.push({ d, j });
        if (api.onIngredient) api.onIngredient(i, j, true);
      } else if (!want && has) {
        const it = d.spawned.get(j);
        if (it) physics.remove(it);
        d.spawned.delete(j);
        queue = queue.filter(q => !(q.d === d && q.j === j));
        if (api.onIngredient) api.onIngredient(i, j, false);
      }
    }
    d.liquid.setLevel(MAX_FILL * Math.min(1, Math.max(0, (x - 0.5) / n)));
    d.liquid.setTint(Math.min(1, p * 1.1));
    const eff = x >= n + 0.4;
    if (eff !== d.effectOn) { d.effectOn = eff; runEffect(d, eff); }
  };

  api.tick = function (dt) {
    if (cur < 0) return;
    const d = dr[cur];
    acc += dt;
    if (queue.length && acc >= SPAWN_GAP) {
      acc = 0;
      const q = queue.shift();
      if (q.d === d && d.spawned.has(q.j)) {
        const ing = d.c.ingredients[q.j];
        d.spawned.set(q.j, spawn(d, ing.piece, d.c.color, q.j));
      }
    }
    for (let k = streams.length - 1; k >= 0; k--) {
      const s = streams[k];
      s.v -= 120 * dt;
      s.m.position.y += s.v * dt;
      const sy = s.d.liquid.surfaceY();
      if (s.m.position.y <= sy + 0.1 || s.m.position.y < 0) {
        s.d.liquid.ripple(s.m.position.x, s.m.position.z, 0.45);
        s.m.parent && s.m.parent.remove(s.m);
        s.m.material.dispose();
        streams.splice(k, 1);
      }
    }
    d.liquid.update(dt);
    dr.forEach((o, k) => { if (k !== cur && o.glass.group.visible) o.liquid.update(dt); });
  };

  api.nudge = function () {
    if (cur < 0) return;
    physics.nudge();
    dr[cur].liquid.ripple(0, 0, 1.1);
  };
  api.activeGlass = () => dr[cur >= 0 ? cur : 0].glass;
  api.surfaceY = () => (cur >= 0 ? dr[cur].liquid.surfaceY() : 0);
  api.rippleActive = (x, z, s) => { if (cur >= 0) dr[cur].liquid.ripple(x, z, s); };
  return api;
}
```
